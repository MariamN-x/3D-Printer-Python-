add component -type Python -name InputComponent
add component -type Python -name ProcessorComponent

add port -componentName InputComponent -name inPort -gateway python2DtEthernet
add port -componentName ProcessorComponent -name procPort -gateway python2DtEthernet

config port -componentName InputComponent -portName inPort -macAddress 12:34:56:78:90:01 -ipAddress 192.168.2.10
config port -componentName ProcessorComponent -portName procPort -macAddress 12:34:56:78:90:02 -ipAddress 192.168.2.11

# ---- transport between the two ports ----
add portSocket -componentName InputComponent -portName inPort -ipPortNum 9000 -socketType client -protocol tcpIp
add portSocket -componentName ProcessorComponent -portName procPort -ipPortNum 9000 -socketType server -protocol tcpIp

define componentSignals -componentName InputComponent -signals [gcode_line:uint8_t[512]:output,gcode_len:uint32_t:output,startSignal:uint8_t:output,refillCommand:uint8_t:output]

define componentSignals -componentName ProcessorComponent -signals [gcode_line:uint8_t[512]:input,gcode_len:uint32_t:input,startSignal:uint8_t:input,refillCommand:uint8_t:input,headX:double:output,headY:double:output,headZ:double:output,nozzleTemp:double:output,bedTemp:double:output,filamentRemaining:double:output,printerState:int:output,errorCode:int:output]

connect signals -sourceSignal InputComponent.gcode_line -destSignal ProcessorComponent.gcode_line -sourcePortName inPort -destPortName procPort
connect signals -sourceSignal InputComponent.gcode_len -destSignal ProcessorComponent.gcode_len -sourcePortName inPort -destPortName procPort
connect signals -sourceSignal InputComponent.startSignal -destSignal ProcessorComponent.startSignal -sourcePortName inPort -destPortName procPort
connect signals -sourceSignal InputComponent.refillCommand -destSignal ProcessorComponent.refillCommand -sourcePortName inPort -destPortName procPort

set workspaceDir .
set digitalTwinName printerTwin
set simulationStep 0.01 s
set totalSimTime 600 s

generate
